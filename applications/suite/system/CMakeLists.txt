cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(KyronextSystem VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_AUTOMOC ON)
  include(GNUInstallDirs)
  include(CTest)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
  add_subdirectory(../ui ui-framework)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Network)
find_package(KF6KIO REQUIRED)
add_library(KyronextMetrics STATIC metrics.cpp metrics.h)
target_link_libraries(KyronextMetrics PUBLIC Qt6::Core Qt6::Network)
target_include_directories(KyronextMetrics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(KyronextFileOperations STATIC fileops.cpp fileops.h)
target_link_libraries(KyronextFileOperations PUBLIC Qt6::Core KF6::KIOCore)
target_include_directories(KyronextFileOperations PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(kyronext-monitor monitor.cpp)
target_link_libraries(kyronext-monitor PRIVATE KyronextUi KyronextMetrics)
add_executable(kyronext-files files.cpp)
target_link_libraries(kyronext-files PRIVATE KyronextUi KyronextFileOperations KF6::KIOWidgets)
install(TARGETS kyronext-monitor kyronext-files RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES resources/kyronext-monitor.desktop resources/kyronext-files.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES resources/kyronext-monitor.svg resources/kyronext-files.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-system tests/system_test.cpp)
  target_link_libraries(test-system PRIVATE KyronextMetrics KyronextFileOperations Qt6::Test)
  add_test(NAME system-metrics-and-files COMMAND test-system)
  set_tests_properties(system-metrics-and-files PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
  add_executable(test-monitor-render tests/monitor_capture.cpp)
  target_link_libraries(test-monitor-render PRIVATE KyronextUi KyronextMetrics)
  add_test(NAME monitor-offscreen-render COMMAND test-monitor-render
           "${CMAKE_CURRENT_BINARY_DIR}/kyronext-monitor-preview.png")
  set_tests_properties(monitor-offscreen-render PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
    TIMEOUT 15)
endif()
