find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia)
add_library(KyronextAmpPlaylist STATIC playlist.cpp playlist.h)
target_link_libraries(KyronextAmpPlaylist PUBLIC Qt6::Core)
target_include_directories(KyronextAmpPlaylist PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(kyronext-amp main.cpp processed_audio.cpp processed_audio.h visualizer.h dsp.h)
target_compile_features(kyronext-amp PRIVATE cxx_std_17)
target_link_libraries(kyronext-amp PRIVATE KyronextUi KyronextAmpPlaylist Qt6::Widgets Qt6::Multimedia)
qt_add_resources(kyronext-amp amp-art PREFIX "/amp" FILES resources/kyronext-amp.svg)
install(TARGETS kyronext-amp RUNTIME DESTINATION bin)
install(FILES resources/kyronext-amp.desktop DESTINATION share/applications)
install(FILES resources/kyronext-amp.svg DESTINATION share/icons/hicolor/scalable/apps)
if(BUILD_TESTING)

  target_link_libraries(test-amp PRIVATE KyronextAmpPlaylist Qt6::Test Qt6::Multimedia)
  add_test(NAME amp-playlist-and-decoder COMMAND test-amp)
  set_tests_properties(amp-playlist-and-decoder PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(test-amp-dsp tests/test_dsp.cpp processed_audio.cpp processed_audio.h dsp.h)
  target_include_directories(test-amp-dsp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(test-amp-dsp PRIVATE Qt6::Test Qt6::Multimedia)
  add_test(NAME amp-real-dsp COMMAND test-amp-dsp)
  set_tests_properties(amp-real-dsp PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(test-amp-ui tests/test_ui.cpp processed_audio.cpp processed_audio.h visualizer.h)

  qt_add_resources(test-amp-ui amp-ui-art PREFIX "/amp" FILES resources/kyronext-amp.svg)
  add_test(NAME amp-ui-controls COMMAND test-amp-ui)
  set_tests_properties(amp-ui-controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
