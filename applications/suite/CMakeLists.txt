cmake_minimum_required(VERSION 3.21)
project(KyronextSuite VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
add_subdirectory(ui)
add_subdirectory(settings)
add_subdirectory(update)
option(KYRONEXT_BUILD_BACKUP "Build Kyronext Backup" ON)
if(KYRONEXT_BUILD_BACKUP)
  add_subdirectory(backup)
endif()
option(KYRONEXT_BUILD_DISK_HEALTH "Build read-only Kyronext Disk Health" ON)
if(KYRONEXT_BUILD_DISK_HEALTH)
  add_subdirectory(disk-health)
endif()
option(KYRONEXT_BUILD_PAINT "Build Kyronext Paint" ON)
option(KYRONEXT_BUILD_AMP "Build Kyronext AMP" ON)
if(KYRONEXT_BUILD_PAINT)
  add_subdirectory(paint)
endif()

# Paquet local expérimental : ne modifie jamais le système pendant la compilation.
set(CPACK_PACKAGE_NAME "kyronext-suite-preview")
set(CPACK_PACKAGE_VENDOR "Kyronext Systems - By Manix")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Applications natives Kyronext en validation")
set(CPACK_PACKAGE_CONTACT "Kyronext Systems <on3egs@users.noreply.github.com>")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_program(KYRONEXT_DPKG_SHLIBDEPS dpkg-shlibdeps REQUIRED)
  set(CPACK_GENERATOR "DEB;TGZ")
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
  set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
  set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
  set(CPACK_DEBIAN_PACKAGE_RELEASE "0~preview3")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
if(KYRONEXT_BUILD_AMP AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/amp/CMakeLists.txt")
  add_subdirectory(amp)
endif()
option(KYRONEXT_BUILD_SYSTEM "Build Kyronext Files and Monitor" ON)
if(KYRONEXT_BUILD_SYSTEM AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/system/CMakeLists.txt")
  add_subdirectory(system)
endif()
option(KYRONEXT_BUILD_ACCESSORIES "Build Kyronext media accessories" ON)
if(KYRONEXT_BUILD_ACCESSORIES AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/accessories/CMakeLists.txt")
  add_subdirectory(accessories)
endif()
option(KYRONEXT_BUILD_ORGANIZER "Build Archive, Calendar and Clock" ON)
if(KYRONEXT_BUILD_ORGANIZER AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/organizer/CMakeLists.txt")
  add_subdirectory(organizer)
endif()
option(KYRONEXT_BUILD_NAVIGATOR "Build experimental browser (not a default browser)" ON)
if(KYRONEXT_BUILD_NAVIGATOR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/navigator/CMakeLists.txt")
  add_subdirectory(navigator)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # Dépendances runtime par module optionnel. Les bibliothèques C++ directes
  # sont aussi vérifiées par dpkg-shlibdeps ; ceci couvre les outils externes
  # et les intégrations Qt/KDE utilisées à l'exécution.
  set(_kyronext_deb_dependencies libqt6widgets6)
  if(KYRONEXT_BUILD_BACKUP)
    list(APPEND _kyronext_deb_dependencies tar)
  endif()
  if(KYRONEXT_BUILD_SYSTEM)
    list(APPEND _kyronext_deb_dependencies kio6)
  endif()
  if(KYRONEXT_BUILD_ORGANIZER)
    list(APPEND _kyronext_deb_dependencies libkf6archive6)
  endif()
  if(KYRONEXT_BUILD_AMP)
    list(APPEND _kyronext_deb_dependencies libqt6multimedia6)
  endif()
  if(KYRONEXT_BUILD_ACCESSORIES)
    list(APPEND _kyronext_deb_dependencies libqt6multimediawidgets6)
  endif()
  if(KYRONEXT_BUILD_NAVIGATOR)
    list(APPEND _kyronext_deb_dependencies libqt6webenginewidgets6)
  endif()
  list(REMOVE_DUPLICATES _kyronext_deb_dependencies)
  string(JOIN ", " CPACK_DEBIAN_PACKAGE_DEPENDS ${_kyronext_deb_dependencies})
endif()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_subdirectory(tests)
endif()
include(CPack)
