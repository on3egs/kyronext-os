target_compile_features(kyronext-navigator PRIVATE cxx_std_17)
target_link_libraries(kyronext-navigator PRIVATE KyronextUi Qt6::WebEngineWidgets)

    add_executable(test-navigator-engine tests/test_engine.cpp browser.cpp browser.h secure_page.h policy.h)

    add_test(NAME navigator-engine-local COMMAND test-navigator-engine)
    set_tests_properties(navigator-engine-local PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_OPENGL=software;QT_QUICK_BACKEND=software;LIBGL_ALWAYS_SOFTWARE=1" TIMEOUT 60)
endif()
