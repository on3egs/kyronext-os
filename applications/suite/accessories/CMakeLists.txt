find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets)
foreach(app gallery video capture)
  add_executable(kyronext-${app} ${app}.cpp)
  target_compile_features(kyronext-${app} PRIVATE cxx_std_17)
  target_link_libraries(kyronext-${app} PRIVATE KyronextUi Qt6::Widgets)
  install(TARGETS kyronext-${app} RUNTIME DESTINATION bin)
  install(FILES resources/kyronext-${app}.desktop DESTINATION share/applications)
  install(FILES resources/kyronext-${app}.svg DESTINATION share/icons/hicolor/scalable/apps)
endforeach()
target_link_libraries(kyronext-video PRIVATE Qt6::Multimedia Qt6::MultimediaWidgets)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-accessory-images tests/image_test.cpp)
  target_link_libraries(test-accessory-images PRIVATE Qt6::Widgets Qt6::Test)
  add_test(NAME accessory-image-files COMMAND test-accessory-images)
  set_tests_properties(accessory-image-files PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  add_executable(test-accessory-ui tests/ui_test.cpp)
  target_link_libraries(test-accessory-ui PRIVATE KyronextUi Qt6::Widgets Qt6::Test Qt6::Multimedia Qt6::MultimediaWidgets)
  add_test(NAME accessory-ui-and-permissions COMMAND test-accessory-ui)
  set_tests_properties(accessory-ui-and-permissions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
