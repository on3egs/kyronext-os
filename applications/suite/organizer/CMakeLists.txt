cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(KyronextOrganizer VERSION 0.1.0 LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_AUTOMOC ON)
  include(GNUInstallDirs)
  include(CTest)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
  add_subdirectory(../ui ui-framework)
endif()
find_package(KF6Archive REQUIRED)
add_library(KyronextOrganizerData STATIC archive_backend.cpp calendar_store.cpp)
target_link_libraries(KyronextOrganizerData PUBLIC Qt6::Core KF6::Archive)
target_include_directories(KyronextOrganizerData PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
foreach(app archive calendar clock)
  add_executable(kyronext-${app} ${app}.cpp)
  target_link_libraries(kyronext-${app} PRIVATE KyronextUi KyronextOrganizerData)
  install(TARGETS kyronext-${app} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES resources/kyronext-${app}.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES resources/kyronext-${app}.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
endforeach()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(test-organizer tests/organizer_test.cpp)
  target_link_libraries(test-organizer PRIVATE KyronextOrganizerData Qt6::Test)
  add_test(NAME organizer-archives-and-events COMMAND test-organizer)
  set_tests_properties(organizer-archives-and-events PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
  add_executable(test-clock tests/clock_test.cpp)
  target_include_directories(test-clock PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(test-clock PRIVATE KyronextUi Qt6::Test)
  add_test(NAME organizer-clock-controls COMMAND test-clock)
  set_tests_properties(organizer-clock-controls PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
endif()
